41 bool slide(SlidingTilesEnums::Direction direction) {
42 using namespace SlidingTilesData;
43 bool canSlide = false;
44 //The main purpose of the below check is to ensure the board is not empty, meaning the default board is the current board.
45 if (board.hasEmptyRow()) return false;
46
47 //Determine if a slide can occur in the desired direction
48 switch (direction) {
49 case SlidingTilesEnums::Direction::DOWN:
50 canSlide = board.canAccess(currentRow + 1, currentColumn);
51 break;
52 case SlidingTilesEnums::Direction::UP:
53 canSlide = board.canAccess(currentRow - 1, currentColumn);
54 break;
55 case SlidingTilesEnums::Direction::LEFT:
56 canSlide = board.canAccess(currentRow, currentColumn - 1);
57 break;
58 case SlidingTilesEnums::Direction::RIGHT:
59 canSlide = board.canAccess(currentRow, currentColumn + 1);
60 break;
61 }
62
63 if (!canSlide) return false;
64
65 //Slide the desired square to the empty square
66 switch (direction) {
67 case SlidingTilesEnums::Direction::DOWN:
68 std::swap(board.access(currentRow, currentColumn), board.access(currentRow + 1, currentColumn));
69 currentRow += 1;
70 break;
71 case SlidingTilesEnums::Direction::UP:
72 std::swap(board.access(currentRow, currentColumn), board.access(currentRow - 1, currentColumn));
73 currentRow -= 1;
74 break;
75 case SlidingTilesEnums::Direction::LEFT:
76 std::swap(board.access(currentRow, currentColumn), board.access(currentRow, currentColumn - 1));
77 currentColumn -= 1;
78 break;
79 case SlidingTilesEnums::Direction::RIGHT:
80 std::swap(board.access(currentRow, currentColumn), board.access(currentRow, currentColumn + 1));
81 currentColumn += 1;
82 break;
83 }
84 return true;
85 }
